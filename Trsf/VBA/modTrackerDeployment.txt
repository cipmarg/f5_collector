Option Explicit


'===========================================================
' CONFIGURATION
'===========================================================

Private Const DEPLOY_SHEET As String = "Tracker Deployment"
Private Const DEPLOY_TABLE As String = "TrackerDeploy"

Private Const SUMMARY_SHEET As String = "Summary"

Private Const MODE_TEMPLATE As String = "TEMPLATE"
Private Const MODE_DEPLOYED As String = "DEPLOYED"

Private Const OUTPUT_PREFIX As String = "Tech_Tracker_"

Private Const CALC_TIMEOUT_SECONDS As Long = 120


'===========================================================
' MAIN DEPLOYMENT PROCEDURE
'===========================================================

Public Sub DeployTrackers()

    Dim wsDeploy As Worksheet
    Dim tbl As ListObject
    Dim tblRow As ListRow

    Dim basePath As String
    Dim packageName As String
    Dim overwriteExisting As Boolean

    Dim outcome As String
    Dim outputFile As String

    Dim originalMode As Variant
    Dim originalDeployVisibility As XlSheetVisibility

    Dim previousScreenUpdating As Boolean
    Dim previousDisplayAlerts As Boolean
    Dim previousEnableEvents As Boolean
    Dim previousStatusBar As Variant

    Dim autoSaveWasOn As Boolean
    Dim autoSaveStateKnown As Boolean

    Dim colPackage As Long
    Dim colDeploy As Long
    Dim colOverwrite As Long
    Dim colStatus As Long
    Dim colLastDeployed As Long
    Dim colOutputFile As Long

    Dim selectedCount As Long
    Dim processedCount As Long

    Dim createdCount As Long
    Dim replacedCount As Long
    Dim skippedCount As Long
    Dim failedCount As Long

    Dim fatalNumber As Long
    Dim fatalDescription As String

    Dim setupComplete As Boolean


    On Error GoTo FatalError


    '-------------------------------------------------------
    ' Locate deployment table
    '-------------------------------------------------------

    Set wsDeploy = _
        ThisWorkbook.Worksheets(DEPLOY_SHEET)

    Set tbl = _
        wsDeploy.ListObjects(DEPLOY_TABLE)


    If tbl.DataBodyRange Is Nothing Then

        MsgBox _
            "The TrackerDeploy table is empty.", _
            vbInformation, _
            "Deploy Trackers"

        Exit Sub

    End If


    '-------------------------------------------------------
    ' Resolve table columns by header
    '-------------------------------------------------------

    colPackage = _
        tbl.ListColumns("Migration Package").Index

    colDeploy = _
        tbl.ListColumns("Deploy Tracker").Index

    colOverwrite = _
        tbl.ListColumns("Overwrite Existing").Index

    colStatus = _
        tbl.ListColumns("Status").Index

    colLastDeployed = _
        tbl.ListColumns("Last Deployed").Index

    colOutputFile = _
        tbl.ListColumns("Output File").Index


    '-------------------------------------------------------
    ' Read workbook configuration
    '-------------------------------------------------------

    basePath = Trim$(CStr( _
        ThisWorkbook.Names("Tracker_Base_Path") _
                    .RefersToRange.Value2 _
    ))


    If Len(basePath) = 0 Then

        Err.Raise _
            vbObjectError + 2001, , _
            "Tracker_Base_Path is empty."

    End If


    originalMode = _
        ThisWorkbook.Names("Tracker_Mode") _
                    .RefersToRange.Value2


    originalDeployVisibility = _
        wsDeploy.Visible


    'Generated trackers must not be allowed to deploy trackers.
    If UCase$(Trim$(CStr(originalMode))) <> MODE_TEMPLATE Then

        Err.Raise _
            vbObjectError + 2002, , _
            "Tracker deployment can only be run from the template."

    End If


    '-------------------------------------------------------
    ' Count rows selected for deployment
    '-------------------------------------------------------

    For Each tblRow In tbl.ListRows

        If IsYes( _
            tblRow.Range.Cells(1, colDeploy).Value2 _
        ) Then

            selectedCount = selectedCount + 1

        End If

    Next tblRow


    If selectedCount = 0 Then

        MsgBox _
            "No rows have Deploy Tracker set to YES.", _
            vbInformation, _
            "Deploy Trackers"

        Exit Sub

    End If


    '-------------------------------------------------------
    ' Preserve Excel state
    '-------------------------------------------------------

    previousScreenUpdating = _
        Application.ScreenUpdating

    previousDisplayAlerts = _
        Application.DisplayAlerts

    previousEnableEvents = _
        Application.EnableEvents

    previousStatusBar = _
        Application.StatusBar


    Application.ScreenUpdating = False
    Application.DisplayAlerts = False

    'Required because changing Migration_Package must fire
    'the existing Worksheet_Change code.
    Application.EnableEvents = True


    '-------------------------------------------------------
    ' Ensure VBA can update protected Summary cells
    '-------------------------------------------------------

    SetProgress _
        "Preparing deployment: enabling VBA access to Summary..."

    EnsureSummaryVBAAccess


    '-------------------------------------------------------
    ' Temporarily disable AutoSave
    '-------------------------------------------------------

    On Error Resume Next

    autoSaveWasOn = ThisWorkbook.AutoSaveOn

    If Err.Number = 0 Then

        autoSaveStateKnown = True

        ThisWorkbook.AutoSaveOn = False

    End If

    Err.Clear

    On Error GoTo FatalError


    setupComplete = True


    '-------------------------------------------------------
    ' Process selected packages
    '-------------------------------------------------------

    For Each tblRow In tbl.ListRows


        If IsYes( _
            tblRow.Range.Cells(1, colDeploy).Value2 _
        ) Then


            processedCount = processedCount + 1


            packageName = Trim$(CStr( _
                tblRow.Range.Cells( _
                    1, _
                    colPackage _
                ).Value2 _
            ))


            overwriteExisting = IsYes( _
                tblRow.Range.Cells( _
                    1, _
                    colOverwrite _
                ).Value2 _
            )


            SetProgress _
                "Deploying tracker " & _
                processedCount & _
                " of " & _
                selectedCount & _
                ": " & _
                packageName


            tblRow.Range.Cells( _
                1, _
                colStatus _
            ).Value2 = "Processing..."


            outputFile = vbNullString


            '------------------------------------------------
            ' Deploy one tracker
            '------------------------------------------------

            If Len(packageName) = 0 Then

                outcome = _
                    "ERROR: Migration Package is blank"

            Else

                outcome = DeployOneTracker( _
                    packageName:=packageName, _
                    basePath:=basePath, _
                    overwriteExisting:=overwriteExisting, _
                    restoreVisibility:=originalDeployVisibility, _
                    outputFile:=outputFile _
                )

            End If


            '------------------------------------------------
            ' Record result
            '------------------------------------------------

            tblRow.Range.Cells( _
                1, _
                colStatus _
            ).Value2 = outcome


            If Len(outputFile) > 0 Then

                tblRow.Range.Cells( _
                    1, _
                    colOutputFile _
                ).Value2 = outputFile

            End If


            Select Case outcome

                Case "Created"

                    createdCount = createdCount + 1

                    tblRow.Range.Cells( _
                        1, _
                        colLastDeployed _
                    ).Value = Now


                Case "Replaced"

                    replacedCount = replacedCount + 1

                    tblRow.Range.Cells( _
                        1, _
                        colLastDeployed _
                    ).Value = Now


                Case "Already exists - skipped"

                    skippedCount = skippedCount + 1


                Case Else

                    If UCase$( _
                        Left$( _
                            Trim$(outcome), _
                            6 _
                        ) _
                    ) = "ERROR:" Then

                        failedCount = failedCount + 1

                    End If

            End Select


            SetProgress _
                packageName & _
                ": " & _
                outcome


        End If

    Next tblRow


CleanUp:

    '-------------------------------------------------------
    ' Restore safe template state
    '
    'Migration_Package is intentionally NOT restored.
    'The master remains on the last package processed.
    '-------------------------------------------------------

    On Error Resume Next


    If setupComplete Then

        SetProgress _
            "Finalising deployment: restoring template state..."

        RestoreTemplateState _
            originalDeployVisibility, _
            vbNullString

    End If


    '-------------------------------------------------------
    ' Restore AutoSave
    '-------------------------------------------------------

    If autoSaveStateKnown Then

        ThisWorkbook.AutoSaveOn = _
            autoSaveWasOn

    End If


    '-------------------------------------------------------
    ' Restore Excel application state
    '-------------------------------------------------------

    Application.DisplayAlerts = _
        previousDisplayAlerts

    Application.ScreenUpdating = _
        previousScreenUpdating

    Application.EnableEvents = _
        previousEnableEvents

    Application.StatusBar = _
        previousStatusBar


    On Error GoTo 0


    '-------------------------------------------------------
    ' Final report
    '-------------------------------------------------------

    If fatalNumber <> 0 Then

        MsgBox _
            "Tracker deployment stopped." & _
            vbCrLf & vbCrLf & _
            "Error " & _
            fatalNumber & _
            ": " & _
            fatalDescription, _
            vbExclamation, _
            "Deploy Trackers"

    Else

        MsgBox _
            "Tracker deployment finished." & _
            vbCrLf & vbCrLf & _
            "Created: " & createdCount & vbCrLf & _
            "Replaced: " & replacedCount & vbCrLf & _
            "Skipped: " & skippedCount & vbCrLf & _
            "Failed: " & failedCount, _
            vbInformation, _
            "Deploy Trackers"

    End If


    Exit Sub


FatalError:

    fatalNumber = Err.Number
    fatalDescription = Err.Description

    Resume CleanUp

End Sub


'===========================================================
' DEPLOY ONE TRACKER
'===========================================================

Private Function DeployOneTracker( _
    ByVal packageName As String, _
    ByVal basePath As String, _
    ByVal overwriteExisting As Boolean, _
    ByVal restoreVisibility As XlSheetVisibility, _
    ByRef outputFile As String _
) As String


    Dim fso As Object

    Dim destinationFolder As String
    Dim destinationFile As String

    Dim tempFolder As String
    Dim tempFile As String

    Dim existingFile As Boolean

    Dim validationError As String
    Dim calculationError As String

    Dim deploymentStatePrepared As Boolean

    Dim errorNumber As Long
    Dim errorDescription As String


    On Error GoTo Failed


    Set fso = _
        CreateObject("Scripting.FileSystemObject")


    '-------------------------------------------------------
    ' Build destination paths
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": checking destination folder..."


    destinationFolder = _
        CombinePath( _
            basePath, _
            packageName _
        )


    destinationFile = _
        CombinePath( _
            destinationFolder, _
            OUTPUT_PREFIX & _
            SafeFilePart(packageName) & _
            ".xlsm" _
        )


    outputFile = destinationFile


    '-------------------------------------------------------
    ' Destination folder must already exist
    '-------------------------------------------------------

    If Not fso.FolderExists(destinationFolder) Then

        DeployOneTracker = _
            "ERROR: Folder not found"

        Exit Function

    End If


    existingFile = _
        fso.FileExists(destinationFile)


    '-------------------------------------------------------
    ' Existing-file policy
    '-------------------------------------------------------

    If existingFile And _
       Not overwriteExisting Then

        DeployOneTracker = _
            "Already exists - skipped"

        Exit Function

    End If


    '-------------------------------------------------------
    ' Set Migration Package
    '
    'Events MUST remain enabled here.
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": setting Migration Package..."


    Application.EnableEvents = True


    ThisWorkbook.Names("Migration_Package") _
                .RefersToRange.Value2 = _
                packageName


    DoEvents


    '-------------------------------------------------------
    ' Wait for workbook calculation
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": calculating workbook..."


    If Not WaitForWorkbookReady( _
        CALC_TIMEOUT_SECONDS, _
        calculationError _
    ) Then

        DeployOneTracker = _
            "ERROR: " & calculationError

        Exit Function

    End If


    '-------------------------------------------------------
    ' Validate resulting tracker data
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": validating tracker..."


    validationError = _
        ValidateTrackerState(packageName)


    If Len(validationError) > 0 Then

        DeployOneTracker = _
            "ERROR: " & validationError

        Exit Function

    End If


    '-------------------------------------------------------
    ' Prepare the state stored in the generated copy
    '
    'Set flag BEFORE calling the routine, so cleanup will
    'restore the master even if preparation fails halfway.
    '-------------------------------------------------------

    deploymentStatePrepared = True


    PrepareDeployedState _
        packageName


    '-------------------------------------------------------
    ' Create unique temporary filename
    '-------------------------------------------------------

    tempFolder = _
        GetTempFolder()


    tempFile = _
        CombinePath( _
            tempFolder, _
            "~" & _
            OUTPUT_PREFIX & _
            SafeFilePart(packageName) & _
            "_" & _
            Format$(Now, "yyyymmdd_hhnnss") & _
            "_" & _
            CStr(CLng(Timer * 100)) & _
            ".xlsm" _
        )


    '-------------------------------------------------------
    ' Save local copy
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": creating local tracker copy..."


    ThisWorkbook.SaveCopyAs _
        tempFile


    '-------------------------------------------------------
    ' Restore open template immediately
    '-------------------------------------------------------

    RestoreTemplateState _
        restoreVisibility, _
        packageName


    deploymentStatePrepared = False


    '-------------------------------------------------------
    ' Copy local file into OneDrive migration folder
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": copying tracker to migration folder..."


    fso.CopyFile _
        tempFile, _
        destinationFile, _
        overwriteExisting


    '-------------------------------------------------------
    ' Remove temporary file
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": cleaning temporary file..."


    fso.DeleteFile _
        tempFile, _
        True


    tempFile = vbNullString


    '-------------------------------------------------------
    ' Result
    '-------------------------------------------------------

    If existingFile Then

        DeployOneTracker = "Replaced"

    Else

        DeployOneTracker = "Created"

    End If


    SetProgress _
        packageName & _
        ": deployment complete"


CleanUp:

    '-------------------------------------------------------
    ' Always restore template after a partial deployment
    '-------------------------------------------------------

    On Error Resume Next


    If deploymentStatePrepared Then

        RestoreTemplateState _
            restoreVisibility, _
            packageName

    End If


    'Remove a leftover TEMP copy after errors.
    If Not fso Is Nothing Then

        If Len(tempFile) > 0 Then

            If fso.FileExists(tempFile) Then

                fso.DeleteFile _
                    tempFile, _
                    True

            End If

        End If

    End If


    On Error GoTo 0


    If errorNumber <> 0 Then

        DeployOneTracker = _
            "ERROR: " & _
            errorNumber & _
            " - " & _
            errorDescription

    End If


    Exit Function


Failed:

    errorNumber = Err.Number
    errorDescription = Err.Description

    Resume CleanUp

End Function


'===========================================================
' ENSURE VBA ACCESS TO PROTECTED SUMMARY
'===========================================================

Private Sub EnsureSummaryVBAAccess()

    Dim ws As Worksheet
    Dim sheetPass As String


    sheetPass = _
        GetSheetPassword()


    If Len(sheetPass) = 0 Then

        Err.Raise _
            vbObjectError + 2200, , _
            "Sheet password configuration is empty."

    End If


    Set ws = _
        ThisWorkbook.Worksheets(SUMMARY_SHEET)


    SetProgress _
        "Preparing deployment: applying UserInterfaceOnly protection..."


    'This keeps the sheet protected for users while allowing
    'VBA to write to locked cells, including Migration_Package.
    ws.Protect _
        Password:=sheetPass, _
        UserInterfaceOnly:=True, _
        AllowFiltering:=True, _
        AllowSorting:=True


    sheetPass = vbNullString
    Set ws = Nothing

End Sub


'===========================================================
' SINGLE SOURCE FOR SHEET PASSWORD
'===========================================================

Private Function GetSheetPassword() As String

    'Keep CFG_A local here.
    'This avoids the module-level constant issue encountered
    'previously, while giving all deployment routines one
    'single password source.

    Const CFG_A As String = "$$44pt33hstt"


    GetSheetPassword = _
        PasswordDecode(CFG_A)

End Function


'===========================================================
' PREPARE STATE STORED IN GENERATED TRACKER
'===========================================================

Private Sub PrepareDeployedState( _
    ByVal packageName As String _
)

    Dim previousEvents As Boolean

    Dim errorNumber As Long
    Dim errorDescription As String


    previousEvents = _
        Application.EnableEvents


    On Error GoTo Failed


    Application.EnableEvents = False


    '-------------------------------------------------------
    ' Mark workbook as deployed
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": setting DEPLOYED mode..."


    ThisWorkbook.Names("Tracker_Mode") _
                .RefersToRange.Value2 = _
                MODE_DEPLOYED


    '-------------------------------------------------------
    ' Switch away from deployment sheet
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": switching to Summary..."


    ThisWorkbook.Worksheets(SUMMARY_SHEET) _
                .Activate


    '-------------------------------------------------------
    ' Hide deployment administration sheet
    '-------------------------------------------------------

    SetProgress _
        packageName & _
        ": hiding deployment sheet..."


    ThisWorkbook.Worksheets(DEPLOY_SHEET) _
                .Visible = _
                xlSheetVeryHidden


    SetProgress _
        packageName & _
        ": deployed state ready"


CleanUp:

    Application.EnableEvents = _
        previousEvents


    'Disable this handler before propagating an error.
    On Error GoTo 0


    If errorNumber <> 0 Then

        Err.Raise _
            errorNumber, _
            "PrepareDeployedState", _
            errorDescription

    End If


    Exit Sub


Failed:

    errorNumber = Err.Number
    errorDescription = Err.Description

    Resume CleanUp

End Sub


'===========================================================
' RESTORE OPEN MASTER TEMPLATE
'===========================================================

Private Sub RestoreTemplateState( _
    ByVal deploySheetVisibility As XlSheetVisibility, _
    ByVal packageName As String _
)

    Dim previousEvents As Boolean


    previousEvents = _
        Application.EnableEvents


    On Error GoTo CleanUp


    Application.EnableEvents = False


    If Len(packageName) > 0 Then

        SetProgress _
            packageName & _
            ": restoring template state..."

    End If


    '-------------------------------------------------------
    ' Restore deployment sheet visibility
    '-------------------------------------------------------

    ThisWorkbook.Worksheets(DEPLOY_SHEET) _
                .Visible = _
                deploySheetVisibility


    '-------------------------------------------------------
    ' Restore template marker
    '-------------------------------------------------------

    ThisWorkbook.Names("Tracker_Mode") _
                .RefersToRange.Value2 = _
                MODE_TEMPLATE


CleanUp:

    Application.EnableEvents = _
        previousEvents

End Sub


'===========================================================
' WAIT FOR CALCULATION
'===========================================================

Private Function WaitForWorkbookReady( _
    ByVal timeoutSeconds As Long, _
    ByRef errorDetail As String _
) As Boolean


    Dim startTime As Double
    Dim elapsed As Double


    On Error GoTo Failed


    startTime = Timer


    'Calculate dirty cells in all open workbooks.
    Application.Calculate


    Do While _
        Application.CalculationState <> xlDone


        DoEvents


        elapsed = _
            Timer - startTime


        'Timer resets at midnight.
        If elapsed < 0 Then

            elapsed = _
                elapsed + 86400

        End If


        If elapsed >= timeoutSeconds Then

            errorDetail = _
                "Calculation timeout after " & _
                timeoutSeconds & _
                " seconds"

            Exit Function

        End If

    Loop


    DoEvents


    WaitForWorkbookReady = True

    Exit Function


Failed:

    errorDetail = _
        "Calculation error " & _
        Err.Number & _
        ": " & _
        Err.Description

End Function


'===========================================================
' VALIDATE TRACKER BEFORE COPYING
'===========================================================

Private Function ValidateTrackerState( _
    ByVal expectedPackage As String _
) As String


    Dim valueToCheck As Variant


    On Error GoTo Failed


    '-------------------------------------------------------
    ' Migration Package
    '-------------------------------------------------------

    valueToCheck = _
        ThisWorkbook.Names("Migration_Package") _
                    .RefersToRange.Value2


    If IsError(valueToCheck) Then

        ValidateTrackerState = _
            "Migration Package contains an error"

        Exit Function

    End If


    If StrComp( _
        Trim$(CStr(valueToCheck)), _
        expectedPackage, _
        vbTextCompare _
    ) <> 0 Then

        ValidateTrackerState = _
            "Migration Package did not update correctly"

        Exit Function

    End If


    '-------------------------------------------------------
    ' Migration Type
    '-------------------------------------------------------

    valueToCheck = _
        ThisWorkbook.Names("Migration_Type") _
                    .RefersToRange.Value2


    If IsError(valueToCheck) Then

        ValidateTrackerState = _
            "Migration Type contains an error"

        Exit Function

    End If


    If Len(Trim$(CStr(valueToCheck))) = 0 Then

        ValidateTrackerState = _
            "Migration Type is empty"

        Exit Function

    End If


    '-------------------------------------------------------
    ' Prep Start Date
    '-------------------------------------------------------

    valueToCheck = _
        ThisWorkbook.Names("Prep_Start_Date") _
                    .RefersToRange.Value


    If IsError(valueToCheck) Or _
       Not IsDate(valueToCheck) Then

        ValidateTrackerState = _
            "Prep Start Date is invalid"

        Exit Function

    End If


    '-------------------------------------------------------
    ' Migration Date
    '-------------------------------------------------------

    valueToCheck = _
        ThisWorkbook.Names("Migration_Date") _
                    .RefersToRange.Value


    If IsError(valueToCheck) Or _
       Not IsDate(valueToCheck) Then

        ValidateTrackerState = _
            "Migration Date is invalid"

        Exit Function

    End If


    Exit Function


Failed:

    ValidateTrackerState = _
        "Validation error " & _
        Err.Number & _
        ": " & _
        Err.Description

End Function


'===========================================================
' STATUS BAR HELPER
'===========================================================

Private Sub SetProgress( _
    ByVal messageText As String _
)

    Application.StatusBar = _
        messageText

    DoEvents

End Sub


'===========================================================
' YES / NO HELPER
'===========================================================

Private Function IsYes( _
    ByVal valueToCheck As Variant _
) As Boolean


    IsYes = _
        UCase$(Trim$(CStr(valueToCheck))) = "YES"

End Function


'===========================================================
' TEMP FOLDER
'===========================================================

Private Function GetTempFolder() As String

    Dim tempPath As String


    tempPath = _
        Environ$("TEMP")


    If Len(tempPath) = 0 Then

        tempPath = _
            Environ$("TMP")

    End If


    If Len(tempPath) = 0 Then

        tempPath = _
            ThisWorkbook.Path

    End If


    GetTempFolder = _
        tempPath

End Function


'===========================================================
' PATH HELPER
'===========================================================

Private Function CombinePath( _
    ByVal parentPath As String, _
    ByVal childPath As String _
) As String


    If Right$( _
        parentPath, _
        1 _
    ) = Application.PathSeparator Then

        CombinePath = _
            parentPath & childPath

    Else

        CombinePath = _
            parentPath & _
            Application.PathSeparator & _
            childPath

    End If

End Function


'===========================================================
' FILE-NAME SANITISER
'===========================================================

Private Function SafeFilePart( _
    ByVal textValue As String _
) As String


    Dim invalidCharacters As Variant
    Dim item As Variant


    invalidCharacters = Array( _
        "\", _
        "/", _
        ":", _
        "*", _
        "?", _
        """", _
        "<", _
        ">", _
        "|" _
    )


    SafeFilePart = _
        Trim$(textValue)


    For Each item In invalidCharacters

        SafeFilePart = _
            Replace( _
                SafeFilePart, _
                CStr(item), _
                "_" _
            )

    Next item

End Function
